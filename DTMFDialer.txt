/*
DTMF Dialer for PicoCalc PicoMite BASIC
by Christopher Fox
*/

' Create the DTMF tones array.
Dim tones(16,2)
For i = 0 To 3
tones(i,0)=697
tones(i+4,0)=770
tones(i+8,0)=852
tones(i+12,0)=941
Next i
For i = 0 To 15 Step 4
tones(i,1)=1209
tones(i+1,1)=1336
tones(i+2,1)=1477
tones(i+3,1)=1633
Next i

' Draw the screen.
CONST FG_Color = RGB(green)
CONST BG_Color = RGB(Black)
CONST T_BG_Color = RGB(64,64,64)

' Reserve memory for GUI controls.
CONST Caption = 1
CONST Num_1 = 2
CONST Num_2 = 3
CONST Num_3 = 4
Const Let_A = 5
Const Num_4 = 6
Const Num_5 = 7
Const Num_6 = 8
Const Let_B = 9
Const Num_7 = 10
Const Num_8 = 11
Const Num_9 = 12
Const Let_C = 13
Const Star = 14
Const Num_0 = 15
Const Hash = 16
Const Let_D = 17

GUI CAPTION Caption,"Dialpad",mm.hres/2,10,"CMN",FG_Color,BG_Color

' Row 1: 1 2 3 A
GUI DISPLAYBOX Num_1,10,10,75,75,FG_Color,BG_Color
CtrlVal(Num_1) = "1"
GUI DISPLAYBOX Num_2,85,10,75,75,FG_Color,BG_Color
CtrlVal(Num_2) = "2"
GUI DISPLAYBOX Num_3,160,10,75,75,FG_Color,BG_Color
CtrlVal(Num_3) = "3"
GUI DISPLAYBOX Let_A,235,10,75,75,FG_Color,BG_Color
CtrlVal(Let_A) = "A"

' Row 2: 4 5 6 B
GUI DISPLAYBOX Num_4,10,85,75,75,FG_Color,BG_Color
CtrlVal(Num_4) = "4"
GUI DISPLAYBOX Num_5,85,85,75,75,FG_Color,BG_Color
CtrlVal(Num_5) = "5"
GUI DISPLAYBOX Num_6,160,85,75,75,FG_Color,BG_Color
CtrlVal(Num_6) = "6"
GUI DISPLAYBOX Let_B,235,85,75,75,FG_Color,BG_Color
CtrlVal(Let_B) = "B"

' Row 3: 7 8 9 C
GUI DISPLAYBOX Num_7,10,160,75,75,FG_Color,BG_Color
CtrlVal(Num_7) = "7"
GUI DISPLAYBOX Num_8,85,160,75,75,FG_Color,BG_Color
CtrlVal(Num_8) = "8"
GUI DISPLAYBOX Num_9,160,160,75,75,FG_Color,BG_Color
CtrlVal(Num_9) = "9"
GUI DISPLAYBOX Let_C,235,160,75,75,FG_Color,BG_Color
CtrlVal(Let_C) = "C"

' Row 4: * 0 # D
GUI DISPLAYBOX Star,10,235,75,75,FG_Color,BG_Color
CtrlVal(Star) = "*"
GUI DISPLAYBOX Num_0,85,235,75,75,FG_Color,BG_Color
CtrlVal(Num_0) = "0"
GUI DISPLAYBOX Hash,160,235,75,75,FG_Color,BG_Color
CtrlVal(Hash) = "#"
GUI DISPLAYBOX Let_D,235,235,75,75,FG_Color,BG_Color
CtrlVal(Let_D) = "D"

' Main Loop Wait until keypress then end.
DO
Pressed_key$ = inkey$
IF Pressed_key$="q" then cls:gui delete all:end
IF Pressed_key$="1" then GUI BCOLOUR T_BG_Color,Num_1:play tone tones(0,0),tones(0,1),100:pause 100:GUI BCOLOUR BG_Color,Num_1
IF Pressed_key$="2" then GUI BCOLOUR T_BG_Color,Num_2:play tone tones(1,0),tones(1,1),100:pause 100:GUI BCOLOUR BG_Color,Num_2
IF Pressed_key$="3" then GUI BCOLOUR T_BG_Color,Num_3:play tone tones(2,0),tones(2,1),100:pause 100:GUI BCOLOUR BG_Color,Num_3
IF Pressed_key$="a" then GUI BCOLOUR T_BG_Color,Let_A:play tone tones(3,0),tones(3,1),100:pause 100:GUI BCOLOUR BG_Color,Let_A
IF Pressed_key$="4" then GUI BCOLOUR T_BG_Color,Num_4:play tone tones(4,0),tones(4,1),100:pause 100:GUI BCOLOUR BG_Color,Num_4
IF Pressed_key$="5" then GUI BCOLOUR T_BG_Color,Num_5:play tone tones(5,0),tones(5,1),100:pause 100:GUI BCOLOUR BG_Color,Num_5
IF Pressed_key$="6" then GUI BCOLOUR T_BG_Color,Num_6:play tone tones(6,0),tones(6,1),100:pause 100:GUI BCOLOUR BG_Color,Num_6
IF Pressed_key$="b" then GUI BCOLOUR T_BG_Color,Let_B:play tone tones(7,0),tones(7,1),100:pause 100:GUI BCOLOUR BG_Color,Let_B
IF Pressed_key$="7" then GUI BCOLOUR T_BG_Color,Num_7:play tone tones(8,0),tones(8,1),100:pause 100:GUI BCOLOUR BG_Color,Num_7
IF Pressed_key$="8" then GUI BCOLOUR T_BG_Color,Num_8:play tone tones(9,0),tones(9,1),100:pause 100:GUI BCOLOUR BG_Color,Num_8
IF Pressed_key$="9" then GUI BCOLOUR T_BG_Color,Num_9:play tone tones(10,0),tones(10,1),100:pause 100:GUI BCOLOUR BG_Color,Num_9
IF Pressed_key$="c" then GUI BCOLOUR T_BG_Color,Let_C:play tone tones(11,0),tones(11,1),100:pause 100:GUI BCOLOUR BG_Color,Let_C
IF Pressed_key$="*" then GUI BCOLOUR T_BG_Color,Star:play tone tones(12,0),tones(12,1),100:pause 100:GUI BCOLOUR BG_Color,Star
IF Pressed_key$="0" then GUI BCOLOUR T_BG_Color,Num_0:play tone tones(13,0),tones(13,1),100:pause 100:GUI BCOLOUR BG_Color,Num_0
IF Pressed_key$="#" then GUI BCOLOUR T_BG_Color,Hash:play tone tones(14,0),tones(14,1),100:pause 100:GUI BCOLOUR BG_Color,Hash
IF Pressed_key$="d" then GUI BCOLOUR T_BG_Color,Let_D:play tone tones(15,0),tones(15,1),100:pause 100:GUI BCOLOUR BG_Color,Let_D
Loop
